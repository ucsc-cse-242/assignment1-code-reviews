from random import sample
import numpy as np
import matplotlib.pyplot as plt

mean = [1, -1]
covariance_mat = [[3, 2], [2, 2]]

# sample a multivariate normal distribution
samples = np.random.multivariate_normal(mean, covariance_mat, 100)

x = samples[:, 0]
y = samples[:, 1]

plt.scatter(x, y)
# plt.show()

# center dataset
samples_centered = samples - mean

# compute covarience matrix
centered_covar_mat = np.cov(samples.T)

# find eigen values and vectors
eigen_values, eigen_vectors = np.linalg.eigh(centered_covar_mat)

# select PCA vectors
pca1 = eigen_vectors[:, -1]
pca2 = eigen_vectors[:, -2]

# Q4, compute varience projected onto pca 1 and 2
pca1_variance = eigen_values[-1]
pca2_variance = eigen_values[-2]
print(pca1_variance, pca2_variance)
